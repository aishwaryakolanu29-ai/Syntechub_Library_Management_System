import java.util.Scanner;
import java.io.IOException;

public class Main {

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);
        Library library = new Library();

        // Load previously saved books
        try {
            library.loadBooks();
            System.out.println("Library data loaded successfully!");
        } catch (IOException e) {
            System.out.println("Error loading library data.");
        }

        int choice;

        do {
            System.out.println("\n================================");
            System.out.println("    LIBRARY MANAGEMENT SYSTEM");
            System.out.println("================================");
            System.out.println("1. Add Book");
            System.out.println("2. Remove Book");
            System.out.println("3. Search Book");
            System.out.println("4. Display All Books");
            System.out.println("5. Exit");
            System.out.println("================================");

            System.out.print("Enter your choice: ");

            try {
                choice = Integer.parseInt(sc.nextLine().trim());

            } catch (NumberFormatException e) {
                System.out.println("Please enter a valid number!");
                continue;
            }

            switch (choice) {

                case 1:
                    System.out.print("Enter Book ID: ");

                    try {
                        int id = Integer.parseInt(sc.nextLine().trim());

                        if (id <= 0) {
                            System.out.println(
                                    "Book ID must be positive!");
                            break;
                        }

                        System.out.print("Enter Book Name: ");
                        String name = sc.nextLine().trim();

                        System.out.print("Enter Author Name: ");
                        String author = sc.nextLine().trim();

                        if (name.isEmpty() || author.isEmpty()
                                || name.contains("|")
                                || author.contains("|")) {
                            System.out.println(
                                    "Invalid book or author name!");
                            break;
                        }

                        Book book = new Book(id, name, author);

                        if (library.addBook(book)) {
                            library.saveBooks();
                        }

                    } catch (NumberFormatException e) {
                        System.out.println("Invalid Book ID!");
                    } catch (IOException e) {
                        System.out.println("Error saving book!");
                    }
                    break;

                case 2:
                    System.out.print("Enter Book ID to remove: ");

                    try {
                        int removeId = Integer.parseInt(
                                sc.nextLine().trim());

                        if (library.removeBook(removeId)) {
                            library.saveBooks();
                        }

                    } catch (NumberFormatException e) {
                        System.out.println("Invalid Book ID!");
                    } catch (IOException e) {
                        System.out.println("Error updating file!");
                    }
                    break;

                case 3:
                    System.out.print("Enter Book ID to search: ");

                    try {
                        int searchId = Integer.parseInt(
                                sc.nextLine().trim());

                        library.searchBook(searchId);

                    } catch (NumberFormatException e) {
                        System.out.println("Invalid Book ID!");
                    }
                    break;

                case 4:
                    library.displayBooks();
                    break;

                case 5:
                    System.out.println(
                            "Thank you for using our library system!");
                    break;

                default:
                    System.out.println(
                            "Invalid choice! Please select 1 to 5.");
            }

        } while (choice != 5);

        sc.close();
    }
}
